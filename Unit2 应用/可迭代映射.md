# 可迭代映射

对应英文原页：https://solidity-by-example.org/app/iterable-mapping

Solidity 中的 `mapping` 默认无法遍历。本示例展示如何构建一个可迭代的映射——通过维护一个键数组和辅助映射来实现遍历，同时保持 O(1) 的查找效率。

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.26;

// IterableMapping 库：实现从 address 到 uint256 的可迭代映射
library IterableMapping {
    // Map 结构体：可迭代映射的核心数据结构
    struct Map {
        address[] keys;                       // 存储所有键的数组（用于迭代）
        mapping(address => uint256) values;   // 键到值的映射
        mapping(address => uint256) indexOf;  // 键到数组索引的映射
        mapping(address => bool) inserted;    // 键是否已存在的标记
    }

    // get 函数：根据键获取值
    function get(Map storage map, address key) public view returns (uint256) {
        return map.values[key];
    }

    // getKeyAtIndex 函数：根据数组索引获取键
    function getKeyAtIndex(Map storage map, uint256 index)
        public
        view
        returns (address)
    {
        return map.keys[index];
    }

    // size 函数：获取映射中键值对的数量
    function size(Map storage map) public view returns (uint256) {
        return map.keys.length;
    }

    // set 函数：设置键值对（如果键已存在则更新，不存在则新增）
    function set(Map storage map, address key, uint256 val) public {
        if (map.inserted[key]) {
            // 键已存在：直接更新值
            map.values[key] = val;
        } else {
            // 键不存在：新增
            map.inserted[key] = true;
            map.values[key] = val;
            map.indexOf[key] = map.keys.length;  // 记录键在数组中的索引
            map.keys.push(key);                  // 将键添加到数组末尾
        }
    }

    // remove 函数：删除键值对（使用交换删除法保持数组紧凑）
    function remove(Map storage map, address key) public {
        if (!map.inserted[key]) {
            return;  // 键不存在，直接返回
        }

        delete map.inserted[key];
        delete map.values[key];

        // 交换删除法：用最后一个元素覆盖要删除的元素，然后 pop
        uint256 index = map.indexOf[key];                // 要删除键在数组中的索引
        address lastKey = map.keys[map.keys.length - 1]; // 数组中最后一个键

        map.indexOf[lastKey] = index;  // 更新最后一个键的索引为被删除的位置
        delete map.indexOf[key];       // 清除被删除键的索引记录

        map.keys[index] = lastKey;  // 用最后一个键覆盖要删除的位置
        map.keys.pop();             // 删除数组最后一个元素
    }
}

// TestIterableMap 合约：测试可迭代映射的使用
contract TestIterableMap {
    // using for 语法：将 IterableMapping 库附加到 Map 类型
    using IterableMapping for IterableMapping.Map;

    IterableMapping.Map private map;

    // setInMapping 函数：调用者在映射中设置自己的值
    function setInMapping(uint256 val) public {
        map.set(msg.sender, val);
    }

    // getFromMap 函数：调用者获取自己的值
    function getFromMap() public view returns (uint256) {
        return map.get(msg.sender);
    }

    // getKeyAtIndex 函数：根据索引获取键
    function getKeyAtIndex(uint256 index) public view returns (address) {
        return map.getKeyAtIndex(index);
    }

    // sizeOfMapping 函数：获取映射的大小
    function sizeOfMapping() public view returns (uint256) {
        return map.size();
    }

    // removeFromMapping 函数：调用者从映射中删除自己的条目
    function removeFromMapping() public {
        map.remove(msg.sender);
    }
}
```
