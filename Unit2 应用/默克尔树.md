# 默克尔树

对应英文原页：https://solidity-by-example.org/app/merkle-tree

默克尔树（Merkle Tree）允许你以加密方式证明某个元素存在于一个集合中，而无需透露整个集合的内容。

常用于空投、白名单验证等场景。

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.26;

// MerkleProof 合约：通用的默克尔证明验证工具
contract MerkleProof {
    // verify 函数：验证默克尔证明
    function verify(
        bytes32[] memory proof,  // 证明路径数组（兄弟节点哈希）
        bytes32 root,            // 默克尔树根哈希
        bytes32 leaf,            // 要验证的叶子节点哈希
        uint256 index            // 叶子节点在树中的索引位置
    ) public pure returns (bool) {
        bytes32 hash = leaf;  // 从叶子节点开始计算

        // 遍历证明路径中的每个元素
        for (uint256 i = 0; i < proof.length; i++) {
            bytes32 proofElement = proof[i];  // 当前层的兄弟节点哈希

            // 根据索引的奇偶性决定拼接顺序
            if (index % 2 == 0) {
                // 索引为偶数：hash 在前（左节点）
                hash = keccak256(abi.encodePacked(hash, proofElement));
            } else {
                // 索引为奇数：proofElement 在前（左节点）
                hash = keccak256(abi.encodePacked(proofElement, hash));
            }

            index = index / 2;  // 索引除以 2，向父节点层移动
        }

        return hash == root;
    }
}

// TestMerkleProof 合约：继承 MerkleProof，构建具体的默克尔树示例
contract TestMerkleProof is MerkleProof {
    bytes32[] public hashes;

    // 构造函数：构建一棵默克尔树
    constructor() {
        // 定义 4 笔交易字符串作为叶子数据
        string[4] memory transactions =
            ["alice -> bob", "bob -> dave", "carol -> alice", "dave -> bob"];

        // 第一步：计算所有叶子节点的哈希值（keccak256）
        for (uint256 i = 0; i < transactions.length; i++) {
            hashes.push(keccak256(abi.encodePacked(transactions[i])));
        }

        // 第二步：逐层构建内部节点
        // 从叶子层开始，每次取相邻两个节点计算父节点
        uint256 n = transactions.length;  // 当前层的节点数（初始为叶子数）
        uint256 offset = 0;               // 当前层在 hashes 数组中的起始位置

        while (n > 0) {
            for (uint256 i = 0; i < n - 1; i += 2) {
                hashes.push(
                    keccak256(
                        abi.encodePacked(
                            hashes[offset + i],      // 左子节点
                            hashes[offset + i + 1]   // 右子节点
                        )
                    )
                );
            }
            offset += n;  // offset 移到下一层
            n = n / 2;    // 每层节点数减半
        }
    }

    // getRoot 函数：获取默克尔树的根哈希
    function getRoot() public view returns (bytes32) {
        return hashes[hashes.length - 1];  // 最后一个元素即为根哈希
    }

    // 验证示例（第 3 个叶子节点，索引为 2）：
    // 叶子哈希：
    //   0xdca3326ad7e8121bf9cf9c12333e6b2271abe823ec9edfe42f813b1e768fa57b
    // 根哈希：
    //   0xcc086fcc038189b4641db2cc4f1de3bb132aefbd65d510d817591550937818c7
    // 证明路径（兄弟节点哈希）：
    //   0x8da9e1c820f9dbd1589fd6585872bc1063588625729e7ab0797cfc63a00bd950
    //   0x995788ffc103b987ad50f5e5707fd094419eb12d9552cc423bd0cd86a3861433
}
```
