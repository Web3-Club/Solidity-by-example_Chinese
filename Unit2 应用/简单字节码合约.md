# 简单字节码合约

对应英文原页：https://solidity-by-example.org/app/simple-bytecode-contract

用字节码编写的合约的简单示例

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.26;

contract Factory {
    event Log(address addr);

    // 部署一个始终返回 255 的合约
    function deploy() external {
        bytes memory bytecode = hex"6960ff60005260206000f3600052600a6016f3";
        address addr;
        assembly {
            // create(value, offset, size)
            addr := create(0, add(bytecode, 0x20), 0x13)
        }
        require(addr != address(0));

        emit Log(addr);
    }
}

interface IContract {
    function getValue() external view returns (uint256);
}

// https://www.evm.codes/playground
/*
运行时代码 - 返回 255
60ff60005260206000f3

// 将 255 存入内存
mstore(p, v) - 将 v 存储到内存 p 到 p + 32

PUSH1 0xff
PUSH1 0
MSTORE

// 从内存返回 32 字节
return(p, s) - 结束执行并从内存 p 到 p + s 返回数据

PUSH1 0x20
PUSH1 0
RETURN

创建代码 - 返回运行时代码
6960ff60005260206000f3600052600a6016f3

// 将运行时代码存入内存
PUSH10 0X60ff60005260206000f3
PUSH1 0
MSTORE

// 从偏移 22 开始返回内存中的 10 字节
PUSH1 0x0a
PUSH1 0x16
RETURN
*/
```
