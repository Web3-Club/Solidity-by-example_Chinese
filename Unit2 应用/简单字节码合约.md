# 简单字节码合约

对应英文原页：https://solidity-by-example.org/app/simple-bytecode-contract

以下是一个完全使用字节码编写的简单合约示例。通过内联汇编的 `create` 操作码部署一个始终返回 `255` 的合约。

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.26;

contract Factory {
    event Log(address addr);

    // deploy 函数：部署一个始终返回 255 的字节码合约
    function deploy() external {
        // 完整字节码：创建代码 + 运行时代码（详见文末解析）
        bytes memory bytecode = hex"6960ff60005260206000f3600052600a6016f3";
        address addr;
        assembly {
            // create(value, offset, size)
            //   value  = 0（不发送以太币）
            //   offset = add(bytecode, 0x20)（跳过前 32 字节的长度前缀）
            //   size   = 0x13（字节码大小，19 字节）
            addr := create(0, add(bytecode, 0x20), 0x13)
        }
        require(addr != address(0));

        emit Log(addr);
    }
}

interface IContract {
    function getValue() external view returns (uint256);
}

/*
字节码解析（参考：https://www.evm.codes/playground）：

运行时代码 —— 60ff60005260206000f3（返回 255）

mstore(p, v) —— 将 v 存储到内存 p 到 p + 32
return(p, s) —— 结束执行并从内存 p 到 p + s 返回数据

PUSH1 0xff    // 将 255 压入栈
PUSH1 0       // 内存偏移量 0
MSTORE        // 将 255 存储到内存
PUSH1 0x20    // 返回大小（32 字节）
PUSH1 0       // 返回起始偏移量
RETURN        // 返回 32 字节（其中包含值 255）

创建代码 —— 6960ff60005260206000f3600052600a6016f3（返回运行时代码）

PUSH10 0x60ff60005260206000f3  // 将运行时代码（10 字节）压入栈
PUSH1 0                         // 内存偏移量 0
MSTORE                          // 将运行时代码存储到内存
PUSH1 0x0a                      // 返回大小（10 字节）
PUSH1 0x16                      // 返回起始偏移量（22 字节，0x16）
RETURN                          // 返回运行时代码
*/
```
