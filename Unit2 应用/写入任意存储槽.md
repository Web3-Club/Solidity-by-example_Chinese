# 写入任意存储槽

对应英文原页：https://solidity-by-example.org/app/write-to-any-slot

Solidity 的存储就像长度为 2^256 的数组。
每个槽可以存储 32 字节。

状态变量的声明顺序和类型决定了它们会使用哪些槽。

不过使用汇编，你可以写入任意存储槽。

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.26;

// StorageSlot 库：用于在特定存储槽位读写数据的工具库
library StorageSlot {
    // 将 address 封装在结构体中，使其可以作为 storage 指针传递
    struct AddressSlot {
        address value;
    }

    // getAddressSlot 函数：通过内联汇编将结构体指针映射到指定存储槽位
    function getAddressSlot(bytes32 slot)
        internal
        pure
        returns (AddressSlot storage pointer)
    {
        assembly {
            // 将结构体的存储槽位设置为 slot
            pointer.slot := slot
        }
    }
}

// TestSlot 合约：演示 StorageSlot 库的用法
contract TestSlot {
    bytes32 public constant TEST_SLOT = keccak256("TEST_SLOT");  // EIP-1967 风格

    // write 函数：将地址写入指定槽位
    function write(address _addr) external {
        StorageSlot.AddressSlot storage data =
            StorageSlot.getAddressSlot(TEST_SLOT);
        data.value = _addr;
    }

    // get 函数：从指定槽位读取地址
    function get() external view returns (address) {
        StorageSlot.AddressSlot storage data =
            StorageSlot.getAddressSlot(TEST_SLOT);
        return data.value;
    }
}
```
