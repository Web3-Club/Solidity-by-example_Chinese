# 写入任意存储槽

对应英文原页：https://solidity-by-example.org/app/write-to-any-slot

Solidity 的存储就像长度为 2^256 的数组。
每个槽可以存储 32 字节。

状态变量的声明顺序和类型决定了它们会使用哪些槽。

不过使用汇编，你可以写入任意存储槽。

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.26;

library StorageSlot {
    // 把 address 包进结构体，以便作为 storage 指针传递
    struct AddressSlot {
        address value;
    }

    function getAddressSlot(bytes32 slot)
        internal
        pure
        returns (AddressSlot storage pointer)
    {
        assembly {
            // 获取存储在 slot 处的 AddressSlot 指针
            pointer.slot := slot
        }
    }
}

contract TestSlot {
    bytes32 public constant TEST_SLOT = keccak256("TEST_SLOT");

    function write(address _addr) external {
        StorageSlot.AddressSlot storage data =
            StorageSlot.getAddressSlot(TEST_SLOT);
        data.value = _addr;
    }

    function get() external view returns (address) {
        StorageSlot.AddressSlot storage data =
            StorageSlot.getAddressSlot(TEST_SLOT);
        return data.value;
    }
}
```
