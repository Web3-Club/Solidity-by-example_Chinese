# 汇编错误

对应英文原页：https://solidity-by-example.org/assembly-error

`assembly` 中错误处理的示例

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.26;

contract AssemblyError {
    function yul_revert(uint256 x) public pure {
        assembly {
            // revert(p, s) - 结束执行
            //                回滚状态更改
            //                返回数据 mem[p…(p+s))
            if gt(x, 10) { revert(0, 0) }
        }
    }
}
```
