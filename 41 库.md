# 库

对应英文原页：https://solidity-by-example.org/library

库（library）类似于合约，但不能声明任何状态变量，也不能发送以太币。

如果库中的所有函数都是 internal，则该库会被嵌入到合约中。

否则必须先部署该库，再在部署合约之前将其链接。

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.26;

library Math {
    function sqrt(uint256 y) internal pure returns (uint256 z) {
        if (y > 3) {
            z = y;
            uint256 x = y / 2 + 1;
            while (x < z) {
                z = x;
                x = (y / x + x) / 2;
            }
        } else if (y != 0) {
            z = 1;
        }
        // 否则 z = 0（默认值）
    }
}

contract TestMath {
    function testSquareRoot(uint256 x) public pure returns (uint256) {
        return Math.sqrt(x);
    }
}

// 用于删除指定下标元素并重新整理数组的函数，
// 使得元素之间没有空隙。
library Array {
    function remove(uint256[] storage arr, uint256 index) public {
        // 将最后一个元素移到要删除的位置
        require(arr.length > 0, "Can't remove from empty array");
        arr[index] = arr[arr.length - 1];
        arr.pop();
    }
}

contract TestArray {
    using Array for uint256[];

    uint256[] public arr;

    function testArrayRemove() public {
        for (uint256 i = 0; i < 3; i++) {
            arr.push(i);
        }

        arr.remove(1);

        assert(arr.length == 2);
        assert(arr[0] == 0);
        assert(arr[1] == 2);
    }
}
```
